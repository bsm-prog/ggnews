import urllib.request
import urllib.parse
import xml.etree.ElementTree as ET
import json

def fetch_local_news(region, keywords):
    """
    지역명과 이슈 키워드를 조합하여 구글 뉴스(RSS)에서 최신 보도자료/기사를 수집하는 함수
    """
    # 검색어 조합 (예: "하남시 (예산 OR 조례 OR 민원)")
    keyword_query = " OR ".join(keywords)
    search_query = f"{region} ({keyword_query})"
    
    # URL 인코딩 및 RSS 주소 생성
    encoded_query = urllib.parse.quote(search_query)
    url = f"https://news.google.com/rss/search?q={encoded_query}&hl=ko&gl=KR&ceid=KR:ko"
    
    # 기사 수집 (보안 우회를 위해 User-Agent 추가)
    req = urllib.request.Request(url, headers={'User-Agent': 'Mozilla/5.0'})
    response = urllib.request.urlopen(req)
    xml_data = response.read()
    
    # XML 데이터 파싱
    root = ET.fromstring(xml_data)
    news_list = []
    
    # 기획에서 약속한 '제외 키워드' 
    exclude_keywords = ["동정", "인사발령", "부고"]
    
    for item in root.findall('./channel/item'):
        title = item.find('title').text
        link = item.find('link').text
        pub_date = item.find('pubDate').text
        
        # 1차 필터링: 제목에 제외 키워드가 있으면 건너뜀
        if any(ex_kw in title for ex_kw in exclude_keywords):
            continue
            
        news_list.append({
            "title": title,
            "link": link,
            "date": pub_date
        })
        
        # 대시보드 테스트를 위해 우선 10개만 추출
        if len(news_list) >= 10:
            break
            
    return news_list

# ==========================================
# 🚀 실행 및 테스트 부분
# ==========================================
# 우리가 세팅한 40개 이슈 키워드 중 일부
ISSUE_KEYWORDS = ["지적", "예산", "조례", "민원", "촉구", "대책"]

# "고양시" 지역의 이슈 기사를 수집해 봅니다.
print("🔎 보도자료를 수집 중입니다...\n")
result_news = fetch_local_news("고양시", ISSUE_KEYWORDS)

# 결과 출력 (나중에는 이 데이터가 HTML 대시보드로 쏙 들어갑니다!)
for idx, news in enumerate(result_news, 1):
    print(f"[{idx}] {news['title']}")
    print(f" └─ 일자: {news['date']}")
    print(f" └─ 원문: {news['link']}\n")

# 최종적으로 대시보드와 연동하기 위해 JSON 파일로 저장하는 코드
with open("news_data.json", "w", encoding="utf-8") as f:
    json.dump(result_news, f, ensure_ascii=False, indent=4)
